\chapter{Ακρωνύμια και Συντομογραφίες}
\label{ch:acronyms}

\begin{english}
\begin{description}[style=nextline, leftmargin=3.5cm, labelwidth=3cm]

  % --- Cryptography ---
  \item[AES] Advanced Encryption Standard
  \item[ECC] Elliptic Curve Cryptography
  \item[HQC] Hamming Quasi-Cyclic
  \item[KAT] Known Answer Test
  \item[KEM] Key Encapsulation Mechanism
  \item[NIST] National Institute of Standards and Technology
  \item[PQC] Post-Quantum Cryptography
  \item[RSA] Rivest--Shamir--Adleman
  \item[SHA-3] Secure Hash Algorithm 3 (Keccak)

  % --- Error Correction ---
  \item[RM] Reed--Muller (code)
  \item[RS] Reed--Solomon (code)

  % --- RISC-V Architecture ---
  \item[ISA] Instruction Set Architecture
  \item[RVV] RISC-V Vector Extension (v1.0)
  \item[Zbb] RISC-V Basic Bit-Manipulation Extension
  \item[Zbc] RISC-V Carry-less Multiplication Extension
  \item[Zbkb] RISC-V Bitmanip for Cryptography Extension
  \item[Zvbb] RISC-V Vector Basic Bit-Manipulation Extension
  \item[Zvbc] RISC-V Vector Carry-less Multiplication Extension
  \item[VLEN] Vector Register Length (hardware parameter)
  \item[VLA] Vector-Length Agnostic (programming model)

  % --- Metrics and Tools ---
  \item[CPI] Cycles Per Instruction
  \item[ISS] Instruction Set Simulator
  \item[LTO] Link-Time Optimization

  % --- Algorithms ---
  \item[FFT] Fast Fourier Transform
  \item[GF] Galois Field

\end{description}
\end{english}
