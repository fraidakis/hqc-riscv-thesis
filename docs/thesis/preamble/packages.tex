% =============================================================================
% Package Imports — LuaLaTeX + Greek/English
% =============================================================================

% --- Engine & font setup ---
\usepackage{fontspec}           % OpenType/TrueType font loading
\usepackage{unicode-math}       % Unicode math fonts (must load before math pkgs)

% --- Language ---
\usepackage{polyglossia}        % Multilingual support
\setdefaultlanguage[variant=monotonic]{greek}
\setotherlanguage{english}

% Greek monospace fallback (Fira Mono lacks Greek glyphs)
\newfontfamily\greekfonttt{DejaVu Sans Mono}[Scale=MatchLowercase]

% --- Typography ---
\usepackage[                    % Full microtypographic improvements
  protrusion=true,
  expansion=true,
  tracking=true
]{microtype}
\usepackage{selnolig}           % Suppress cross-morpheme ligatures (LuaLaTeX)
\usepackage{csquotes}           % Context-sensitive quotes (biblatex needs this)

% --- Page layout ---
\usepackage{scrlayer-scrpage}   % KOMA-Script headers/footers
\usepackage{geometry}           % Page geometry

% --- Math ---
\usepackage{amsmath}            % Math environments
% amssymb NOT needed — unicode-math provides all symbols
\usepackage{mathtools}          % Extensions to amsmath

% --- Figures and graphics ---
\usepackage{graphicx}           % Image inclusion
\usepackage{tikz}               % Programmable graphics
\usepackage{pgfplots}           % Plots from data
\pgfplotsset{compat=1.18}
\usepackage{pgfplotstable}      % Import CSV/TSV data directly into LaTeX tables
\usepackage{subcaption}         % Sub-figures

% --- Tables ---
\usepackage{booktabs}           % Professional table rules
\usepackage{longtable}          % Multi-page tables
\usepackage{tabularx}           % Auto-width columns
\usepackage{multirow}           % Multi-row cells
\usepackage{siunitx}            % SI units and number formatting
\usepackage{tabularray}         % Modern table environment
\UseTblrLibrary{booktabs}       % booktabs-style rules in tabularray

% --- Code listings ---
\usepackage{listings}           % Code listings (primary)
% \usepackage{minted}           % Pygments-based (uncomment if preferred)

% --- Algorithms ---
\usepackage[ruled,vlined,linesnumbered]{algorithm2e}

% --- Cross-references and links ---
\usepackage{hyperref}           % Clickable links, PDF metadata
\usepackage[nameinlink,noabbrev]{cleveref}  % Smart cross-references (TL2026+)

% Fix cleveref + polyglossia Greek: cleveref has no Greek language support
% and \cref@language is left undefined with polyglossia. We set it to
% "english" as a fallback (our explicit \crefname defs override the names).
\makeatletter
\def\cref@language{english}
\makeatother

% Greek cross-reference names
\crefname{chapter}{κεφάλαιο}{κεφάλαια}
\Crefname{chapter}{Κεφάλαιο}{Κεφάλαια}
\crefname{section}{ενότητα}{ενότητες}
\Crefname{section}{Ενότητα}{Ενότητες}
\crefname{subsection}{υποενότητα}{υποενότητες}
\Crefname{subsection}{Υποενότητα}{Υποενότητες}
\crefname{figure}{σχήμα}{σχήματα}
\Crefname{figure}{Σχήμα}{Σχήματα}
\crefname{table}{πίνακα}{πίνακες}
\Crefname{table}{Πίνακας}{Πίνακες}
\crefname{equation}{εξίσωση}{εξισώσεις}
\Crefname{equation}{Εξίσωση}{Εξισώσεις}
\crefname{listing}{κώδικα}{κώδικες}
\Crefname{listing}{Κώδικας}{Κώδικες}
\crefname{algorithm}{αλγόριθμο}{αλγόριθμοι}
\Crefname{algorithm}{Αλγόριθμος}{Αλγόριθμοι}
\crefname{appendix}{παράρτημα}{παραρτήματα}
\Crefname{appendix}{Παράρτημα}{Παραρτήματα}
\crefname{page}{σελίδα}{σελίδες}
\Crefname{page}{Σελίδα}{Σελίδες}

% --- Glossaries: first-use acronym expansion ---
% \gls{key}  → "Full Name (ABBR)" on first use, then "ABBR" — no external tool needed
% \Gls{key}  → sentence-start form  |  \glspl{key} → plural
% All definitions live in preamble/glossary.tex
\usepackage[
  acronym,          % dedicated acronym glossary type
  nomain,           % suppress unused main glossary
  nopostdot,        % no trailing period after descriptions
]{glossaries-extra}

% --- Bibliography ---
\usepackage[
  backend=biber,

  sorting=none,
  maxbibnames=99,
  dashed=false,
  doi=true,
  url=false,
  isbn=false,
  eprint=false
]{biblatex}
\addbibresource{bibliography/references.bib}
