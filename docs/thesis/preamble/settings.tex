% --- siunitx: unified number and unit formatting ---

  reset-math-version = false,
  reset-text-family = false,
  reset-text-series = false,
  reset-text-shape  = false,
  range-phrase      = {--},      % \SIrange{1}{4}{\mega\hertz} \to 1--4 MHz
}

% --- pgfplots: global thesis chart style ---
% Tier-aware, colorblind-safe cycle: same color+marker per tier across all plots.
% NOTE: the canonical \pgfplotscreateplotcyclelist{thesistiers} (4-entry) now
% lives in figures/thesis-tikz-styles.tex, which is \input AFTER this file in
% main.tex. The `cycle list name=thesistiers' reference below is a forward
% reference resolved when each axis is drawn, so no definition is needed here.
\pgfplotsset{
  thesis/.style={
    width=0.85\linewidth,
    height=0.52\linewidth,
    grid=major,
    grid style={line width=0.3pt, draw=gray!25},
    tick align=outside,
    tick label style={font=\small},
    label style={font=\small},
    title style={font=\small\bfseries},
    legend style={
      font=\small, draw=gray!40, fill=white!95,
      at={(0.98,0.98)}, anchor=north east
    },
    every axis/.append style={font=\small},
    cycle list name=thesistiers,
  },
  every axis/.append style={thesis},   % apply to all axes by default
}
