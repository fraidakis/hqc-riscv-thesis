% figures/ch6-polymul/vclmul-leaf-dataflow.tex
% Dataflow diagram: vclmul_leaf splat-and-sweep pattern
\begin{tikzpicture}[
    >=stealth,
    node distance=0.8cm and 1.2cm,
    block/.style={draw, rounded corners, minimum width=2.2cm, minimum height=0.7cm,
                  font=\small\ttfamily, fill=blue!8},
    data/.style={draw, minimum width=1.8cm, minimum height=0.6cm,
                 font=\small, fill=gray!10},
    op/.style={draw, rounded corners, minimum width=2.8cm, minimum height=0.6cm,
               font=\small\ttfamily, fill=green!10},
    acc/.style={draw, rounded corners, minimum width=2.8cm, minimum height=0.6cm,
                font=\small\ttfamily, fill=orange!10},
    arr/.style={->, thick},
  ]

  % Input arrays
  \node[data] (a) {$a[0 \ldots 15]$};
  \node[data, right=3.5cm of a] (b) {$b[0 \ldots 15]$};

  % Load operations
  \node[block, below=0.6cm of a] (splat) {vmv.v.x(a[i])};
  \node[block, below=0.6cm of b] (loadb) {vle64(b[0..15])};

  % Multiply operations
  \node[op, below=1.0cm of $(splat)!0.5!(loadb)$, xshift=-1.5cm] (vclmul) {vclmul.vv};
  \node[op, right=0.4cm of vclmul] (vclmulh) {vclmulh.vv};

  % Results
  \node[data, below=0.8cm of vclmul] (lo) {lo$[0 \ldots 15]$};
  \node[data, below=0.8cm of vclmulh] (hi) {hi$[0 \ldots 15]$};

  % Accumulation
  \node[acc, below=0.8cm of $(lo)!0.5!(hi)$, xshift=-1.2cm] (acclo) {r[i+j] $\oplus$= lo};
  \node[acc, right=0.3cm of acclo] (acchi) {r[i+j+1] $\oplus$= hi};

  % Output
  \node[data, below=0.8cm of $(acclo)!0.5!(acchi)$] (out) {$r[0 \ldots 31]$ (unreduced product)};

  % Arrows
  \draw[arr] (a) -- (splat);
  \draw[arr] (b) -- (loadb);
  \draw[arr] (splat) -- ++(0,-0.4) -| (vclmul.north);
  \draw[arr] (splat) -- ++(0,-0.4) -| (vclmulh.north);
  \draw[arr] (loadb) -- ++(0,-0.4) -| (vclmul.north east);
  \draw[arr] (loadb) -- ++(0,-0.4) -| (vclmulh.north east);
  \draw[arr] (vclmul) -- (lo);
  \draw[arr] (vclmulh) -- (hi);
  \draw[arr] (lo) -- ++(0,-0.3) -| (acclo.north);
  \draw[arr] (hi) -- ++(0,-0.3) -| (acchi.north);
  \draw[arr] (acclo) -- ++(0,-0.3) -| (out.north west);
  \draw[arr] (acchi) -- ++(0,-0.3) -| (out.north east);

  % Annotation: iteration count
  \node[draw=none, font=\footnotesize\itshape, text=blue!70!black,
        right=0.3cm of loadb] {$\times 1$ iteration\\(VLMAX $= 16$)};

  % Outer loop annotation
  \node[draw=none, font=\footnotesize\itshape, text=red!70!black,
        left=0.3cm of splat] {$\forall\, i \in [0,16)$};

\end{tikzpicture}
