\begin{tikzpicture}[

  block/.style={rectangle, draw, fill=#1, minimum width=3.0cm,

    rounded corners=2pt, font=\small},
  block/.default={blue!8},
  arrow/.style={->, >=stealth, thick},

]

% Nodes

\node[block=green!12, right=of rm] (rs) {Reed-Solomon Αποκωδικοποιητής (εξωτερικός)};

% Arrows
\draw[arrow] (input) -- (rm) node[lbl] {$n_1 \times n_2$ bits};

\draw[arrow] (rs) -- (output) node[lbl] {$k$ bytes};

% Annotations below
\node[font=\scriptsize, below=0.3cm of rm, text=gray] {bit-level};
\node[font=\scriptsize, below=0.3cm of rs, text=gray] {symbol-level};
