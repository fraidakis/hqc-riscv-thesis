% Build pipeline diagram: Source → CMake → Toolchain → ELF → Spike → instret
\tikzsetnextfilename{ch5-setup-build-pipeline}
\begin{tikzpicture}[
  node distance=1.2cm and 1.6cm,
  box/.style={
    draw, rounded corners=2pt, minimum height=0.9cm, minimum width=2.0cm,
    font=\small\sffamily, align=center, fill=white
  },
  srcbox/.style={box, fill=blue!8, draw=blue!40},
  toolbox/.style={box, fill=orange!8, draw=orange!40},
  outbox/.style={box, fill=green!8, draw=green!40},
  arr/.style={-{Stealth[length=5pt]}, thick},
  lbl/.style={font=\scriptsize\sffamily, midway, above, text=gray!70!black}
]

% Nodes
\node[srcbox] (src) {Source\\\texttt{.c}/\texttt{.h}/\texttt{.S}};
\node[toolbox, right=of src] (cmake) {CMake\\configuration};
\node[toolbox, right=of cmake] (compiler) {GCC 15.2\\cross-compiler};
\node[outbox, right=of compiler] (elf) {ELF binary\\\texttt{.elf}};
\node[toolbox, right=of elf] (spike) {Spike ISS\\execution};
\node[outbox, right=of spike] (instretnode) {\texttt{instret}\\output};

% Arrows
\draw[arr] (src) -- (cmake);
\draw[arr] (cmake) -- node[lbl] {toolchain file} (compiler);
\draw[arr] (compiler) -- node[lbl] {\texttt{-march}} (elf);
\draw[arr] (elf) -- node[lbl] {\texttt{--isa}} (spike);
\draw[arr] (spike) -- (instretnode);

% Flags annotation below compiler
\node[below=0.4cm of compiler, font=\scriptsize\ttfamily, text=gray!60!black, align=center] (flags)
  {-march=rv64gcv\_zbc\_zbb\_...\\-O3 -ffreestanding};

% Toolchain file annotation below cmake
\node[below=1.4cm of cmake, font=\scriptsize\sffamily, text=gray!60!black, align=center] (tcfile)
  {\texttt{riscv-gcc.cmake}};
\draw[-{Stealth[length=4pt]}, dashed, gray!50] (tcfile) -- (cmake);

\end{tikzpicture}
