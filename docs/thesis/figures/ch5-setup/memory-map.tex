% Memory map diagram for §5.6.2
\tikzsetnextfilename{ch5-setup-memory-map}
\begin{tikzpicture}[
  segment/.style={rectangle, draw=black!60, minimum width=5cm, minimum height=0.9cm,
    align=center, font=\small\sffamily},
  label/.style={font=\scriptsize\ttfamily, anchor=west},
  addr/.style={font=\scriptsize\ttfamily, anchor=east},
]

% Memory segments (bottom to top, matching address growth)
\node[segment, fill=thesisblue!15] (text) at (0,0) {\texttt{.text.startup} + \texttt{.text}\\(code)};
\node[segment, fill=green!10, above=0pt of text] (rodata) {\texttt{.rodata}\\(KAT seeds, constants)};
\node[segment, fill=orange!8, above=0pt of rodata] (tohost) {\texttt{.tohost}\\(HTIF, 64B-aligned)};
\node[segment, fill=yellow!10, above=0pt of tohost] (data) {\texttt{.data}\\(initialized globals)};
\node[segment, fill=red!8, above=0pt of data] (bss) {\texttt{.bss}\\(Keccak state, scratch buffers)};
\node[segment, fill=purple!10, minimum height=1.4cm, above=0pt of bss] (stack) {\texttt{.stack} (512\,KB)\\(polymul tmp, call chain)};

% Address labels (left side)
\node[addr] at ([xshift=-0.3cm]text.south west) {\texttt{0x8000\_0000}};
\node[addr] at ([xshift=-0.3cm]stack.north west) {\texttt{\_stack\_top}};
\node[addr] at ([xshift=-0.3cm]bss.south west) {\texttt{\_bss\_start}};
\node[addr] at ([xshift=-0.3cm]bss.north west) {\texttt{\_bss\_end}};

% Annotations (right side)
\node[label] at ([xshift=0.3cm]text.east) {Entry: \texttt{\_start}};
\node[label] at ([xshift=0.3cm]stack.east) {Grows ↓};

% RAM boundary
\draw[thick, dashed, gray] ([xshift=-2.8cm, yshift=-0.3cm]text.south west) --
  ([xshift=3.5cm, yshift=-0.3cm]text.south east)
  node[midway, below, font=\scriptsize\sffamily] {DRAM base (Spike / QEMU virt)};

% Total RAM annotation
\draw[<->, thick, gray!60] ([xshift=3.2cm]text.south east) -- ([xshift=3.2cm]stack.north east)
  node[midway, right, font=\scriptsize\sffamily, text=gray!80] {128\,MiB};

\end{tikzpicture}
