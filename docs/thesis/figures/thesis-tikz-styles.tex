%              decorations.pathreplacing ; pgfplots ; booktabs ;
%              siunitx ; tabularray ; xcolor   (all in packages.tex).

%              \tikzset / \pgfplotsset / \pgfplotscreateplotcyclelist /
%              tabularray specs (+ one guarded \newif). May be \input
%              more than once without error.
%
% Usage contract: in figures USE the semantic tokens below; do NOT
% inline raw colours (blue!50), font sizes (\footnotesize), line widths
% (line width=0.9pt) or per-figure arrow/legend/palette definitions.
%
% Reorganised by CONCERN (was: historical v1/v2 layers — merged, the

%            the table design system; Tol pale tints for table fills.

% =====================================================================
% 2. TYPOGRAPHY SCALE  (one small, consistent set; sans throughout)
% =====================================================================
\tikzset{
  figT/title/.style   ={font=\sffamily\bfseries\small},
  figT/label/.style   ={font=\sffamily\footnotesize},
  figT/annot/.style   ={font=\sffamily\scriptsize, text=figneutral},
  figT/micro/.style   ={font=\sffamily\scriptsize},
  figT/mono/.style    ={font=\ttfamily\footnotesize},          % code / mnemonics
  figT/callout/.style ={font=\sffamily\scriptsize\itshape, text=figneutral},
  figT/phase/.style   ={font=\sffamily\bfseries, text=figaccent},% KeyGen/Encaps/Decaps headers
}


% =====================================================================
% 3. LINE WEIGHTS + ARROWS  (weight encodes hierarchy; one tip family)
% =====================================================================
\tikzset{
  figLW/hair/.style ={line width=0.4pt},
  figLW/base/.style ={line width=0.7pt},
  figLW/emph/.style ={line width=1.1pt},
  % Semantic edges (single arrows.meta Stealth tip everywhere) ---------
  figedge/.style        ={draw=figrule,    figLW/base, -{Stealth[length=4pt,width=4pt]}},
  figedge/accent/.style ={draw=figaccent,   figLW/emph, -{Stealth[length=5pt,width=5pt]}},
  figedge/weak/.style   ={draw=figneutral,  figLW/hair, dashed, -{Stealth[length=3.5pt]}},
  % Leader / annotation lines (NO arrowhead) connect a label to its target.
  % Distinct from figedge/* (which carry a Stealth tip). Kept because the ch1
  % exemplar figures (nist-rounds-flowchart, acceleration-spectrum) use them.
  figleader/.style       ={semithick, draw=figrule},
  figaccentleader/.style ={thick,     draw=figaccent!70},
}
% Make the bare ->/<- shorthand match figedge/* (covers legacy figures
% that used >=stealth). Set once for all tikzpictures.
\tikzset{>={Stealth[length=4pt,width=4pt]}}


% =====================================================================
% 4. BOXES  (built on the tint ramp + figLW; corners/weights uniform)
% =====================================================================
% NOTE: the old `figaccentbox' and `figmilestone'/`figmilestoneaccent'
% are REMOVED — superseded by figbox/focal and figglyph/* below.
\tikzset{

  figbox/focal/.style ={figbox, draw=figaccent,  figLW/emph, fill=figaccentfill},

% =====================================================================
% 5. GLYPHS  (shape backs up colour -> accessibility) + pill + badge

% --- Canonical line/marker cycle (4 tiers; OVERRIDES settings.tex) ----
% Order = Reference, Tier 1, Tier 2, Tier 3. Marks back up colour (a11y).
\pgfplotscreateplotcyclelist{thesistiers}{%
  {figtierRef, mark=o,         thick},%
  {figtier1,   mark=square*,   thick},%
  {figtier2,   mark=triangle*, thick},%
  {figtier3,   mark=*,         thick}%
}

% --- Reusable chart-type styles (build on the `thesis' base axis) ------
% Charts set ONLY: thesisbar|thesisbarlog|thesisstacked|thesisline|
% thesiswaterfall|thesisdualaxis + data. Width/height/grid/fonts/legend
% come for free from `thesis' (preamble/settings.tex).
\pgfplotsset{
  % Grouped bars (linear) ----------------------------------------------
  thesisbar/.style={
    ybar, bar width=10pt,
    ymajorgrids=true, xmajorgrids=false,
    enlarge x limits=0.25,
    x tick label style={font=\sffamily\small},
    legend style={at={(0.5,1.03)}, anchor=south, legend columns=-1,
                  /tikz/every even column/.append style={column sep=8pt}},
  },
  % Grouped bars (log y) -----------------------------------------------
  thesisbarlog/.style={thesisbar, ymode=log, log origin=infty,
    nodes near coords style={font=\sffamily\tiny, rotate=90, anchor=west}},
  % Stacked bars (100%% breakdown) --------------------------------------
  thesisstacked/.style={
    ybar stacked, bar width=28pt, area legend,
    ymajorgrids=true, xmajorgrids=false,
    enlarge x limits=0.2, ymin=0,
    x tick label style={font=\sffamily\small},
    every node near coords/.append style={font=\sffamily\tiny, anchor=south, yshift=-1pt},
    legend style={at={(0.5,-0.22)}, anchor=north, legend columns=-1,
                  /tikz/every even column/.append style={column sep=8pt}},
  },
  % Single-series cumulative waterfall (ISA ablation) ------------------
  thesiswaterfall/.style={
    ybar, bar width=22pt,
    ymajorgrids=true, xmajorgrids=false,
    enlarge x limits=0.18, ymin=0,
    every axis plot/.append style={fill=figthreatfill, draw=figthreat, figLW/base},
    x tick label style={font=\sffamily\small},
    nodes near coords style={font=\sffamily\tiny, /pgf/number format/fixed},
  },
  % Line / sweep plots --------------------------------------------------
  thesisline/.style={
    cycle list name=thesistiers,
    every axis plot/.append style={mark size=2.5pt},
    legend style={at={(0.98,0.98)}, anchor=north east, draw=figrule,
                  fill=white, fill opacity=0.85, text opacity=1},
  },
  % Dual-axis colour coding (left=accent/process, right=threat) --------
  thesisaxisL/.style={ylabel style={text=figprocess}, y tick label style={text=figprocess}},
  thesisaxisR/.style={axis y line*=right, axis x line=none,
                      ylabel style={text=figthreat}, y tick label style={text=figthreat}},
}

% =====================================================================
% 9. TABLE DESIGN SYSTEM   [NEW — review/verify locally]
% =====================================================================
% Rationale: the 32 table figures currently roll plain booktabs+siunitx
% with NO shared header style, NO rule colour, NO zebra and a misnamed
% "heatmap" that has no colour. Below is a tabularray (tblr) based set
% (tabularray is already loaded; it does cell/row colour natively, so
% no colortbl / [table]{xcolor} needed). booktabs rules still apply.
%
% Migration: replace `\begin{tabular}{...}' with `\begin{thesistblr}{...}'
% and drop manual \toprule/\midrule/\bottomrule (handled by hlines spec).
% S-columns from siunitx still work inside tblr via `si()` or `Q[si=...]`.

% --- Rule + header colours -------------------------------------------
\colorlet{figtablerule}{figaccent!55!black}
\colorlet{figtableheadbg}{figaccent!10}
\colorlet{figtablezebra}{black!4}

% --- Reusable thesis table environment --------------------------------
% Bold sans header row on a faint accent wash, accent booktabs-style
% rules, optional zebra via row style. Light, print-safe.
\NewTblrEnviron{thesistblr}
\SetTblrInner[thesistblr]{
  rowsep      = 2pt,
  colsep      = 6pt,
  hline{1,Z}  = {0.8pt, figtablerule},   % top + bottom (booktabs weight)
  hline{2}    = {0.5pt, figtablerule},   % under header
  row{1}      = {font=\sffamily\bfseries, bg=figtableheadbg},
}

% --- Heatmap cell tint (REAL heatmap for the speedup matrix) ----------
% Tol *pale* washes behind black text (the correct choice for tables).
% Use at the START of a cell to tint it by improvement magnitude, e.g.
%   & \heatHigh{184.8} &   ->  pale-green cell behind the number.
\NewDocumentCommand{\heatLow}{m}{\SetCell{bg=tolPaleYellow}#1}
\NewDocumentCommand{\heatMid}{m}{\SetCell{bg=tolPaleCyan}#1}
\NewDocumentCommand{\heatHigh}{m}{\SetCell{bg=tolPaleGreen}#1}

% --- Best-value emphasis (numeric highlight, no colour dependence) ----
\NewDocumentCommand{\best}{m}{\textbf{#1}}

% --- Disassembly / before-after diff cells ----------------------------
% For zba/zbs/keccak-rotation "without vs with" code comparison tables.
\NewDocumentCommand{\diffOld}{m}{\SetCell{bg=tolPaleRed}   \texttt{#1}}
\NewDocumentCommand{\diffNew}{m}{\SetCell{bg=tolPaleGreen} \texttt{#1}}
