  {\sffamily\bfseries\small\color{figthreat} Χωρίς \Zbc{} (shift-and-XOR loop)}\par\smallskip
  \begin{lstlisting}[language={[RISC-V]Assembler}, basicstyle=\ttfamily\scriptsize,
    numbers=left, numberstyle=\tiny\color{figneutral}, frame=single,
    rulecolor=\color{figthreat}, backgroundcolor=\color{figthreatfill}, xleftmargin=1.5em]
  # a in a0, b in a1
  li    t0, 64       # bit counter
  li    t1, 0        # result_lo = 0
  li    t2, 0        # result_hi = 0
  li    t5, 0        # a_hi = 0 (128-bit shift)
.loop:
  andi  t3, a1, 1    # extract LSB of b
  neg   t3, t3       # mask: 0 or -1
  and   t4, a0, t3   # conditional a_lo
  xor   t1, t1, t4   # accumulate lo
  and   t4, t5, t3   # conditional a_hi
  xor   t2, t2, t4   # accumulate hi
  slli  t5, t5, 1    # a_hi << 1
  srli  t6, a0, 63   # carry out of a_lo
  or    t5, t5, t6   # a_hi |= carry
  slli  a0, a0, 1    # a_lo << 1
  srli  a1, a1, 1    # shift b right
  addi  t0, t0, -1   # decrement
  bnez  t0, .loop    # repeat 64x
  \end{lstlisting}
