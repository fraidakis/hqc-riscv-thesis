% figures/clmul-before-after.tex
% Before/after comparison: naive shift-XOR loop vs clmul+clmulh pair
% for 64-bit polynomial multiplication in GF(2)[x].
% Uses minipage layout for robustness (lstlisting inside TikZ nodes is fragile).

\begin{minipage}[t]{0.48\textwidth}
\centering
\textbf{\sffamily Χωρίς \Zbc (shift-and-XOR loop)}
\begin{lstlisting}[language={[RISC-V]Assembler}, basicstyle=\ttfamily\scriptsize,
  numbers=left, numberstyle=\tiny\color{gray}, frame=single,
  backgroundcolor=\color{red!3}, xleftmargin=1.5em]
  # a in a0, b in a1
  li    t0, 64       # bit counter
  li    t1, 0        # result_lo = 0
  li    t2, 0        # result_hi = 0
  li    t5, 0        # a_hi = 0 (128-bit shift)
.loop:
  andi  t3, a1, 1    # extract LSB of b
  neg   t3, t3       # mask: 0 or -1
  and   t4, a0, t3   # conditional a_lo
  xor   t1, t1, t4   # accumulate lo
  and   t4, t5, t3   # conditional a_hi
  xor   t2, t2, t4   # accumulate hi
  slli  t5, t5, 1    # a_hi << 1
  srli  t6, a0, 63   # carry out of a_lo
  or    t5, t5, t6   # a_hi |= carry
  slli  a0, a0, 1    # a_lo << 1
  srli  a1, a1, 1    # shift b right
  addi  t0, t0, -1   # decrement
  bnez  t0, .loop    # repeat 64x
\end{lstlisting}
\smallskip
{\footnotesize $\approx$ \num{832} εντολές / γινόμενο}
\end{minipage}%
\hfill%
\begin{minipage}[t]{0.48\textwidth}
\centering
\textbf{\sffamily Με \Zbc (\clmul{} + \clmulh)}
\begin{lstlisting}[language={[RISC-V]Assembler}, basicstyle=\ttfamily\scriptsize,
  numbers=left, numberstyle=\tiny\color{gray}, frame=single,
  backgroundcolor=\color{green!3}, xleftmargin=1.5em]
  # a in a0, b in a1
  clmul  t1, a0, a1  # result_lo
  clmulh t2, a0, a1  # result_hi
\end{lstlisting}
\vspace{2em}
{\footnotesize \textbf{\num{2}} εντολές / γινόμενο}
