% --- local field-category palette (semantic; built on shared Tol washes) ---
\colorlet{fmtReg}{tolPaleBlue}     % register operands: rs1, rs2, rd
\colorlet{fmtFunct}{tolPaleGreen}  % function selects: funct7, funct3

\pgfmathsetmacro{\totalwidth}{13}    % cm
\pgfmathsetmacro{\bitwidth}{0.40625} % totalwidth / 32

% Helper: field from bit 'start'..'end' (inclusive, 0-indexed from right)
\newcommand{\bitfield}[5]{%
  \pgfmathsetmacro{\xstart}{(31-#3)*\bitwidth}%
  \pgfmathsetmacro{\w}{(#3-#2+1)*\bitwidth}%
  \node[fmtfield, minimum width=\w cm, fill=#4] at (\xstart cm, #1) {\strut #5};%
}
