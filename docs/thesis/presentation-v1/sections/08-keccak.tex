% --- 08 Keccak & decode ------------------------------------------------------
\begin{frame}{Οι δύο εναπομείναντες πυρήνες: \en{Keccak} \& \en{decode} διανυσματοποιούνται}
  \HL{Μετά τον πολ/σμό, \en{Keccak} και \en{decode} κυριαρχούν, και οι δύο επιταχύνονται με \RVV/\Zvbb/\Zvbc.}
  \vspace{0.4em}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \textbf{\en{Keccak}-$f[1600]$}
      \begin{itemize}\small
        \item Άμεση διανυσματοποίηση μίας κατάστασης \con{αντιπαραγωγική}: ανεπαρκής παραλληλία εντός της στοιβάδας
        \item \HL{2-way \en{SoA}}: δύο ανεξάρτητα \en{permutations} παράλληλα
        \item \Zvbb \vror: εξάλειψη της shift+or εξομοίωσης \enspace$\Rightarrow$\enspace \speedup{1.36}
      \end{itemize}
    \end{column}
    \begin{column}{0.50\textwidth}
      \textbf{\en{Reed-Muller/Solomon decode}}
      \begin{itemize}\small
        \item \en{RM} \en{fast Walsh-Hadamard}: διανυσματικά \RVV βήματα
        \item \en{RS} επί του \GF{2^8} με \vclmul (\Zvbc): \en{Berlekamp-Massey}, \en{Forney}, \en{FFT}
        \item Έναντι βαθμωτής \en{ref.} \enspace$\Rightarrow$\enspace \HL{\speedup{14.3}}
      \end{itemize}
    \end{column}
  \end{columns}
  \pdfpcnote{Δύο kernels μετά τον polymul. Keccak: 2-way SoA + Zvbb vror -> x1.36. Decode: RM RVV FWHT + RS Zvbc (BM/Forney/FFT) -> x14.3 vs scalar ref (src: results/reports/ablation/decode_zvbc.md, HQC-128 x14.27).}
\end{frame}
