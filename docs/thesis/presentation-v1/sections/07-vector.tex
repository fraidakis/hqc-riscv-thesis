\begin{frame}{Από τους \BigO{n} βρόχους στον \BigO{n^2} leaf}
  \begin{itemize}
    \item \textbf{Tier 2:} διανυσματοποιεί το \BigO{n} (ανασύνθεση), όχι το \BigO{n^2} leaf
  \end{itemize}
  \HL{Ο \vclmul{} υπολογίζει πολλαπλά $64\times64$ carry-less γινόμενα ανά διάνυσμα.}

  \begin{columns}[T,onlytextwidth]

        \item Σε \VLEN{}=256, \code{LMUL=m4}: \HL{VLMAX $= T = 16$}, ο εσωτερικός $j$-βρόχος συμπτύσσεται σε μία vector iteration ανά $i$

  \pdfpcnote{.vx μορφή: για κάθε i, το scalar a[i] πολλαπλασιάζεται carry-less με ολόκληρο τον πίνακα b[0..n-1]. Σε VLMAX=16=T ο inner j-loop γίνεται μία iteration, μηδενίζοντας τον scalar inner loop.}

\end{frame}
