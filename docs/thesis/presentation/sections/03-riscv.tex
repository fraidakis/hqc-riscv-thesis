      \begin{itemize}

      \begin{block}{Βαθμωτές (σειριακές)}

          \item \Zbb: περιστροφές/λογικές για \en{Keccak}
        \end{itemize}
      \end{block}
      \begin{block}{Διανυσματικές (\RVV)}
        \begin{itemize}
          \item \Zvbc: \vclmul / \vclmulh, διανυσματικός πολ/σμός
          \item \Zvbb: \vror, διανυσματικές λογικές για \en{Keccak}
        \end{itemize}
      \end{block}
