% --- 06 Scalar optimization (Zbc clmul) --------------------------------------
\begin{frame}[fragile]{Tier 1: σειριακή επιτάχυνση με \Zbc}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.5\textwidth}
      \begin{itemize}
        \item Ο σχολικός πολ/σμός bit-bit αντικαθίσταται από \HL{\clmul} 64×64 bit
        \item \HL{Αναδρομή Karatsuba} για τα μεγάλα πολυώνυμα~\cite{Karatsuba1962}
        \item \HL{\en{Assign-on-first-write}}: κατάργηση αρχικοποίησης ακριβού αποθηκευτή ($-\pct{23}$)
        \item Τρέχει σε \HL{κάθε} σύγχρονο RISC-V core (διαθέσιμο σήμερα)
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{exampleblock}{Πυρήνας \GF{2} πολ/σμού}
\begin{lstlisting}
# 64x64 -> 128-bit carryless
clmul   a2, a0, a1   # low half
clmulh  a3, a0, a1   # high half
\end{lstlisting}
      \end{exampleblock}
      \begin{block}{Αποτέλεσμα}
        Πραγματικό \en{silicon} (\en{Banana Pi BPI-F3}): \HL{\speedup{38}} στο \HQC-128 με μόνο τις βαθμωτές επεκτάσεις.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}
